        function initMap() {

            const slots = getSlots();

                if (!slot.isOpen) {
                    btnClass += 'bg-slate-100 border-slate-200 text-slate-400 cursor-not-allowed opacity-60';
                    btn.disabled = true;
                } else if (slot.isTaken) {
                    btnClass += 'bg-slate-200 border-slate-300 text-slate-500 cursor-not-allowed line-through';
                    btn.disabled = true;
                } else if (selectedSlotId === slot.id) {
                    btnClass += 'bg-rose-600 border-rose-600 text-white shadow-md shadow-rose-600/30';
                } else {
                    btnClass += 'bg-white border-slate-300 text-slate-800 hover:border-rose-500 hover:shadow-sm';
                    btn.onclick = () => {
                        selectedSlotId = slot.id;
                        renderBookingSlots();

                btn.innerHTML = `

                return;

            const venue = document.getElementById('booking-venue').value;
            const slots = getSlots();
            const slot = slots.find(s => s.id === selectedSlotId);

            if (!slot || !slot.isOpen || slot.isTaken) {
                showToast('Selected slot is no longer available.', 'error');
                return;

            slot.isTaken = true;
            slot.bookedBy = user.id;

                slotTime: slot.time,
                slotId: slot.id,
                status: 'Pending Verification (2-hr holding period)',
                createdAt: new Date().toISOString()
            };

            saveSessions(sessions);

            closeModal('booking-modal');
            selectedSlotId = null;
            showToast('Booking submitted successfully! 2-hour holding period active until admin approval.');
            router('dashboard');

                            <button onclick="triggerOpenAllSlotsEmail()" class="px-5 py-3 bg-rose-600 hover:bg-rose-500 text-white font-bold rounded-xl text-sm shadow-lg shadow-rose-600/30 transition">
                                <i class="fa-solid fa-envelope-circle-check mr-2"></i> Open Slots & Email Users
                            </button>
                        </div>
                    </div>

                    <!-- Announcement Broadcast Manager -->
                    <div class="bg-white p-6 sm:p-8 rounded-3xl border border-slate-200 shadow-sm space-y-4">

                            <button onclick="saveAdminAnnouncement()" class="px-6 py-3 bg-slate-900 hover:bg-slate-800 text-white font-bold rounded-xl text-sm">Publish & Broadcast</button>

                    <div class="bg-white p-6 sm:p-8 rounded-3xl border border-slate-200 shadow-sm space-y-6">
                        <div>
                            <h2 class="text-xl font-bold text-slate-900">Edit Schedule & Time Slots (Mon-Sat, 9 AM - 6 PM)</h2>
                            <p class="text-xs text-slate-500">Toggle slot availability. Initially greyed out until admin opens them.</p>

        function toggleSlotStatus(slotId) {
            const slots = getSlots();
            const slot = slots.find(s => s.id === slotId);
            if (slot) {
                slot.isOpen = !slot.isOpen;
                saveSlots(slots);
                showToast(`Slot ${slot.day} ${slot.time} is now ${slot.isOpen ? 'Opened' : 'Closed'}`);
                renderDashboard();
            }
        }

        function triggerOpenAllSlotsEmail() {
            const slots = getSlots();
            slots.forEach(s => s.isOpen = true);
            saveSlots(slots);
            showToast('All Mon-Sat slots opened! Email notification sent to all registered users.');
            renderDashboard();
        }

        function saveAdminAnnouncement() {
            const text = document.getElementById('admin-announcement-input').value;
            localStorage.setItem(DB_KEY_ANNOUNCEMENT, text);
            showToast('Announcement posted to main page and broadcasted!');
            updateNavigation();
        }

        function approveBooking(bookingId) {
            const sessions = getSessions();
            const booking = sessions.find(s => s.id === bookingId);
            if (booking) {
                booking.status = 'Approved (Payment Verified)';
                saveSessions(sessions);
                showToast('Booking approved successfully!');
                renderDashboard();
            }
        }

        function deleteBooking(bookingId) {
            let sessions = getSessions();
            sessions = sessions.filter(s => s.id !== bookingId);
            saveSessions(sessions);
            showToast('Booking deleted.');
            renderDashboard();
        }

        function openStudentNotesModal(userId) {
            const users = getUsers();
            const user = users.find(u => u.id === userId);
            if (!user) return;

            const newNote = prompt(`Add class note or booking history note for ${user.name}:`);
            if (newNote) {
                if (!user.notes) user.notes = [];
                user.notes.push(newNote);
                saveUsers(users);
                showToast('Student note added successfully!');
                renderDashboard();
            }
        }

        window.onload = function() {
            initDefaultSlots();
            updateNavigation();
            router('home');
        };
    </script>
